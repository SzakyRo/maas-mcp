# v0.3.0 — Storage Verbs Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add 15 MAAS storage curated verbs (5 reads + 10 destructive, gated on `confirm: bool`) plus `storage_summary` to maasmcp, behind a `confirm: bool = False` safety gate.

**Architecture:** Same `class XxxTool` pattern as v0.2.0. One file per verb under `src/maasmcp/curated/`. Shared `_storage_paths.py` for path templating (single source of truth). All destructive verbs validate path params → check `confirm` → issue MAAS call. Registered through `_common.py`. Tests use the existing `maas_client` fixture (monkeypatch env vars + `async with` + respx context).

**Tech Stack:** Python 3.11+, FastMCP, Pydantic v2, pytest-asyncio, respx for HTTP mocking.

## Global Constraints

These bind every task. **Do not rephrase; quote verbatim where the spec uses the exact value.**

"""Path templating helpers for storage verbs.

Single source of truth for /nodes/{system_id}/blockdevices/... paths so a
typo can't drift across the 10 verbs that touch this URL space. Per v0.3.0
spec global constraint: no inline f-string path construction elsewhere.
"""

from __future__ import annotations


def block_devices_path(system_id: str) -> str:
    return f"/api/2.0/nodes/{system_id}/blockdevices/"


def block_device_path(system_id: str, block_device_id: int) -> str:
    return f"/api/2.0/nodes/{system_id}/blockdevices/{block_device_id}/"


def partitions_path(system_id: str, block_device_id: int) -> str:
    return f"/api/2.0/nodes/{system_id}/blockdevices/{block_device_id}/partitions/"


def partition_path(
    system_id: str, block_device_id: int, partition_id: int
) -> str:
    return (
        f"/api/2.0/nodes/{system_id}/blockdevices/{block_device_id}"
        f"/partitions/{partition_id}"
    )


def set_boot_disk_path(system_id: str, block_device_id: int) -> str:
    return f"/api/2.0/nodes/{system_id}/blockdevices/{block_device_id}/op-set_boot_disk"


def format_block_device_path(system_id: str, block_device_id: int) -> str:
    return f"/api/2.0/nodes/{system_id}/blockdevices/{block_device_id}/op-format"


def format_partition_path(
    system_id: str, block_device_id: int, partition_id: int
) -> str:
    return (
        f"/api/2.0/nodes/{system_id}/blockdevices/{block_device_id}"
        f"/partitions/{partition_id}/op-format"
    )


def mount_partition_path(
    system_id: str, block_device_id: int, partition_id: int
) -> str:
    return (
        f"/api/2.0/nodes/{system_id}/blockdevices/{block_device_id}"
        f"/partitions/{partition_id}/op-mount"

"""list_block_devices: GET /nodes/{system_id}/blockdevices/."""

from __future__ import annotations

from typing import Annotated, Any

from mcp.server.fastmcp import FastMCP
from pydantic.json_schema import SkipJsonSchema

from maasmcp.curated._storage_paths import block_devices_path
from maasmcp.maas.client import MaasClient


class ListBlockDevicesTool:
    name = "list_block_devices"
    description = (
        "List all block devices on a MAAS machine. Returns MAAS's response "
        "(list of dicts). Use get_block_device to drill into one device."
    )

    async def run(
        self,
        client: Annotated[MaasClient, SkipJsonSchema()],
        system_id: str,
    ) -> list[dict[str, Any]]:
        if not system_id:
            raise ValueError("system_id must be a non-empty string")
        resp = await client.get(block_devices_path(system_id))
        if isinstance(resp, dict) and "blockdevices" in resp:
            return list(resp["blockdevices"])
        if isinstance(resp, list):
            return resp
        return []

    def register(self, mcp: FastMCP) -> None:
        mcp.add_tool(self.run, name=self.name, description=self.description)
```

"""get_block_device: GET /nodes/{system_id}/blockdevices/{id}/."""

from __future__ import annotations

from typing import Annotated, Any

from mcp.server.fastmcp import FastMCP
from pydantic.json_schema import SkipJsonSchema

from maasmcp.curated._storage_paths import block_device_path
from maasmcp.maas.client import MaasClient


class GetBlockDeviceTool:
    name = "get_block_device"
    description = (
        "Fetch one block device by system_id + block_device_id. Returns the "
        "MAAS device dict. Use list_block_devices to find IDs first."
    )

    async def run(
        self,
        client: Annotated[MaasClient, SkipJsonSchema()],
        system_id: str,
        block_device_id: int,
    ) -> dict[str, Any]:
        if not system_id:
            raise ValueError("system_id must be a non-empty string")
        if block_device_id <= 0:
            raise ValueError(f"block_device_id must be > 0, got {block_device_id}")
        resp = await client.get(block_device_path(system_id, block_device_id))
        return resp if isinstance(resp, dict) else {}

    def register(self, mcp: FastMCP) -> None:
        mcp.add_tool(self.run, name=self.name, description=self.description)

from __future__ import annotations

from typing import Annotated, Any

from mcp.server.fastmcp import FastMCP
from pydantic.json_schema import SkipJsonSchema

from maasmcp.curated._storage_paths import partitions_path
from maasmcp.maas.client import MaasClient


class CreatePartitionTool:
    name = "create_partition"
    description = (
        "Create a partition on a block device. DESTRUCTIVE — alters the "
        "partition table. Pass confirm=True. Returns the new partition dict."
    )

    async def run(
        self,
        client: Annotated[MaasClient, SkipJsonSchema()],
        system_id: str,
        block_device_id: int,
        size: str,
        fstype: str | None = None,
        partition_table_type: str | None = None,
        bootable: bool | None = None,
        confirm: bool = False,
    ) -> dict[str, Any]:
        if not system_id:
            raise ValueError("system_id must be a non-empty string")
        if block_device_id <= 0:
            raise ValueError(f"block_device_id must be > 0, got {block_device_id}")
        if not size:
            raise ValueError("size must be a non-empty string")
        if confirm is not True:
            raise ValueError(
                "create_partition alters the partition table; pass confirm=True. "
                f"Affected: system_id={system_id}, block_device_id={block_device_id}, "
                f"size={size}"
            )
        body: dict[str, Any] = {"size": size}
        if fstype is not None:
            body["fstype"] = fstype
        if partition_table_type is not None:
            body["partition_table_type"] = partition_table_type
        if bootable is not None:
            body["bootable"] = bootable
        resp = await client.post(
            partitions_path(system_id, block_device_id), json=body
        )
        return resp if isinstance(resp, dict) else {}

    def register(self, mcp: FastMCP) -> None:
        mcp.add_tool(self.run, name=self.name, description=self.description)

from __future__ import annotations

from typing import Annotated, Any

from mcp.server.fastmcp import FastMCP
from pydantic.json_schema import SkipJsonSchema

from maasmcp.curated._storage_paths import mount_partition_path
from maasmcp.maas.client import MaasClient


class MountPartitionTool:
    name = "mount_partition"
    description = (
        "Mount a filesystem at a given mount point. DESTRUCTIVE in the "
        "sense that an agent may pass the wrong mount point and overwrite "
        "or shadow a real filesystem. Unmounting the result can disrupt "
        "running services. Verify the target before passing confirm=True. "
        "Returns {'status': 'ok', ..., 'operation': 'mount'}."
    )

    async def run(
        self,
        client: Annotated[MaasClient, SkipJsonSchema()],
        system_id: str,
        block_device_id: int,
        partition_id: int,
        mount_point: str,
        mount_options: str | None = None,
        confirm: bool = False,
    ) -> dict[str, Any]:
        if not system_id:
            raise ValueError("system_id must be a non-empty string")
        if block_device_id <= 0:
            raise ValueError(f"block_device_id must be > 0, got {block_device_id}")
        if partition_id <= 0:
            raise ValueError(f"partition_id must be > 0, got {partition_id}")
        if not mount_point or not mount_point.startswith("/"):
            raise ValueError(
                f"mount_point must start with '/', got {mount_point!r}"
            )
        if confirm is not True:
            raise ValueError(
                "mount_partition can shadow real filesystems; verify the "
                "target is correct and pass confirm=True. "
                f"Affected: system_id={system_id}, block_device_id={block_device_id}, "
                f"partition_id={partition_id}, mount_point={mount_point}"
            )
        body: dict[str, Any] = {"mount_point": mount_point}
        if mount_options is not None:
            body["mount_options"] = mount_options
        await client.post(
            mount_partition_path(system_id, block_device_id, partition_id), json=body
        )
        return {
            "status": "ok",
            "system_id": system_id,
            "resource_id": partition_id,
            "operation": "mount",
        }

    def register(self, mcp: FastMCP) -> None:

            "resource_id": partition_id,

"""storage_summary: cross-resource read of a machine's storage layout."""

from __future__ import annotations

import asyncio
from typing import Annotated, Any

from mcp.server.fastmcp import FastMCP
from pydantic.json_schema import SkipJsonSchema

from maasmcp.curated._storage_paths import block_devices_path, partitions_path
from maasmcp.maas.client import MaasClient


def _to_gib(size_bytes: int) -> float:
    """Convert bytes (MAAS may return as int or string) to GiB."""
    try:
        n = int(size_bytes)
    except (TypeError, ValueError):
        return 0.0
    return round(n / (1024 ** 3), 2)


class StorageSummaryTool:
    name = "storage_summary"
    description = (
        "Cross-resource read of a machine's storage layout: returns "
        "block_device count, partition count, total size in GiB, plus "
        "raid_devices=0 and volume_groups=0 placeholders (RAID/VG counts "
        "land in v0.4.0). Two-stage fetch: list_block_devices first, "
        "then parallel list_partitions per device."
    )

    async def run(
        self,
        client: Annotated[MaasClient, SkipJsonSchema()],
        system_id: str,
    ) -> dict[str, Any]:
        if not system_id:
            raise ValueError("system_id must be a non-empty string")

        # Stage 1: fetch block devices (sequential — need the count to fan out).
        resp = await client.get(block_devices_path(system_id))
        if isinstance(resp, dict) and "blockdevices" in resp:
            devices = list(resp["blockdevices"])
        elif isinstance(resp, list):
            devices = resp
        else:
            devices = []

        # Stage 2: fetch partitions per device in parallel.
        if devices:
            partition_lists = await asyncio.gather(
                *(client.get(partitions_path(system_id, d["id"])) for d in devices),
                return_exceptions=True,
            )
            partition_count = 0
            for pl in partition_lists:
                if isinstance(pl, Exception):
                    continue
                if isinstance(pl, dict) and "partitions" in pl:
                    partition_count += len(pl["partitions"])
                elif isinstance(pl, list):
                    partition_count += len(pl)
        else:
            partition_count = 0

        total_gib = sum(_to_gib(d.get("size", 0)) for d in devices)

        return {
            "system_id": system_id,
            "block_devices": len(devices),
            "partitions": partition_count,
            "total_size_gb": total_gib,
            "raid_devices": 0,
            "volume_groups": 0,
        }

    def register(self, mcp: FastMCP) -> None:
        mcp.add_tool(self.run, name=self.name, description=self.description)
```
